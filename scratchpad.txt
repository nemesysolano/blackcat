
def generic_predictor(model_name, X_input):
    pattern = re.compile(rf'^{re.escape(model_name)}-\d+-\d+-\d+\.\d+\.keras$')
    model_files = tuple([os.path.join(os.getcwd(), 'models', f) for f in os.listdir(os.path.join(os.getcwd(), 'models')) if pattern.match(f)])
    model_factors = tuple([float(f.split('-')[-1].replace('.keras', '')) for f in model_files])    
    model_importance_sum = np.sum(model_factors)
    model_factors = tuple([f / model_importance_sum for f in model_factors])
    predictions = []

    for index in range(len(model_files)):
        model_file = model_files[index]
        model = tf.keras.models.load_model(model_file)            
        model_predictions = model.predict(X_input, verbose=0)
        predictions.append(model_predictions * model_factors[index])
    predictions = np.sum(predictions, axis=0)
    print(predictions)
    exit(-1)

    return predictions
